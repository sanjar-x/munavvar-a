class BaseService[
    ModelType: BaseModel,
    CreateSchemaType: PydanticSchema,
    UoWType: BaseSQLAlchemyUoW,
](ABC):
    def __init__(self, uow: UoWType):
        self.uow = uow

    @property
    @abstractmethod
    def _repo(self) -> BaseRepository[ModelType]:
        pass

    async def get(self, id: uuid.UUID) -> ModelType | None: ...
    async def get_multi(self, skip=0, limit=100, active_only=True) -> Sequence[ModelType]: ...
    async def add(self, schema: CreateSchemaType) -> ModelType: ...
    async def update[AnyUpdateSchema: PydanticSchema](self, id, schema): ...
    async def archive(self, id: uuid.UUID) -> bool: ...
    async def delete(self, id: uuid.UUID) -> bool: ...
```

import uuid
from dataclasses import dataclass
from datetime import datetime
from typing import Any

from src.modules.catalog.enums import ProductType


@dataclass(frozen=True, slots=True)
class ProductDTO:
    """Frozen data transfer object for Product.

    Field order matches Product model column definitions
    (per D-16).
    """

    id: uuid.UUID
    returnable_item_id: uuid.UUID | None
    type: ProductType
    name: str
    price: int
    attributes: dict[str, Any]
    is_active: bool
    created_at: datetime
    updated_at: datetime


def product_to_dto(product: Any) -> ProductDTO:
    """Convert ORM Product instance to frozen DTO.

    Module-internal converter (per D-10) -- NOT exported
    via public.py. Uses Any type hint to avoid importing
    the ORM model at module level.
    """
    return ProductDTO(
        id=product.id,
        returnable_item_id=product.returnable_item_id,
        type=product.type,
        name=product.name,
        price=product.price,
        attributes=dict(product.attributes),
        is_active=product.is_active,
        created_at=product.created_at,
        updated_at=product.updated_at,
    )
```
